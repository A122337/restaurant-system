menuData = [
    {"id": "m1", "name": "Burger", "price": 25},
    {"id": "m2", "name": "Pizza", "price": 35},
    {"id": "m3", "name": "Juice", "price": 10}
]

tables = [
    {"id": 1, "status": "free"},
    {"id": 2, "status": "free"},
    {"id": 3, "status": "free"}
]

def add_menu_item(name, price):
    new_item = {
        "id": f"m{len(menuData)+1}",
        "name": name,
        "price": price
    }
    menuData.append(new_item)

def delete_menu_item(item_id):
    global menuData
    menuData = [item for item in menuData if item["id"] != item_id]

def toggle_table_status(table_id):
    for table in tables:
        if table["id"] == table_id:
            if table["status"] == "free":
                table["status"] = "occupied"
            else:
                table["status"] = "free"

def notify(message):
    print("Notification:", message)
